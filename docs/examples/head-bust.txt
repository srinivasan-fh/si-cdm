# A head-and-shoulders bust, sculpted from a ball with the brushes only.
#   RUN docs/examples/head-bust.txt        (from the repository folder; about 15 s)
# Millimetres; z up, the face looks along -y. Every stroke with SYMMETRY is
# mirrored across x = 0, so each feature is placed once, on the +x side.
# Points only need to be near the surface: they are snapped onto it.

# The head: a ball, its triangles evened out, then pulled into a skull and a jaw
NEW
SPHERE 85 @0,10,245
SCULPT "Sphere 1" REFINE 6
# even out the ball's triangles
SCULPT "Sphere 1" SMOOTH AT 0,-75,380 0,-75,330 0,-75,280 40,-60,280 40,-60,330 40,-60,380 RADIUS 45 STRENGTH 0.4 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 75,10,380 75,10,330 75,10,280 RADIUS 45 STRENGTH 0.4 SYMMETRY
# narrow the skull, deepen the back, pull the jaw and the chin down out of the ball
SCULPT "Sphere 1" GRAB AT 85,10,330 MOVE -8,0,0 RADIUS 100 SYMMETRY
SCULPT "Sphere 1" GRAB AT 0,95,335 MOVE 0,10,2 RADIUS 100
SCULPT "Sphere 1" GRAB AT 0,-55,265 MOVE 0,-4,-30 RADIUS 90
SCULPT "Sphere 1" GRAB AT 0,-50,240 MOVE 0,-4,-22 RADIUS 70
SCULPT "Sphere 1" GRAB AT 40,-35,245 MOVE -3,-2,-18 RADIUS 55 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,-60,210 30,-45,215 50,-15,230 RADIUS 30 STRENGTH 0.5 SYMMETRY

# The face. The profile: forehead and chin forward
SCULPT "Sphere 1" GRAB AT 0,-60,375 MOVE 0,-8,0 RADIUS 55
SCULPT "Sphere 1" GRAB AT 0,-62,228 MOVE 0,-8,2 RADIUS 32
# jaw corners
SCULPT "Sphere 1" GRAB AT 55,0,245 MOVE 5,0,-8 RADIUS 30 SYMMETRY
# brow ridge, eye sockets, temples
SCULPT "Sphere 1" DRAW AT 8,-77,338 28,-73,341 48,-63,338 RADIUS 13 STRENGTH 0.7 SYMMETRY
SCULPT "Sphere 1" GRAB AT 30,-68,319 MOVE 0,9,0 RADIUS 22 SYMMETRY
SCULPT "Sphere 1" DRAW INVERT AT 66,-30,338 RADIUS 18 STRENGTH 0.15 SYMMETRY
# nose: the ridge, the tip pulled out, the bridge narrowed; wings, tip, nostrils, the crease round the wings
SCULPT "Sphere 1" DRAW AT 0,-78,334 0,-82,318 0,-88,305 RADIUS 12 STRENGTH 1
SCULPT "Sphere 1" GRAB AT 0,-79,292 MOVE 0,-24,-2 RADIUS 26
SCULPT "Sphere 1" SMOOTH AT 0,-95,304 0,-102,292 RADIUS 12 STRENGTH 0.3
SCULPT "Sphere 1" PINCH AT 0,-84,328 0,-100,300 RADIUS 10 STRENGTH 0.3
SCULPT "Sphere 1" GRAB AT 12,-86,282 MOVE 5,-5,0 RADIUS 11 SYMMETRY
SCULPT "Sphere 1" DRAW AT 0,-101,290 RADIUS 9 STRENGTH 0.8
SCULPT "Sphere 1" DRAW INVERT AT 4,-96,281 8,-92,280 RADIUS 3 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" CREASE AT 18,-82,293 21,-81,283 16,-85,276 RADIUS 4 STRENGTH 0.5 SYMMETRY
# mouth: muzzle, lips, the line between them, under the lip, chin
SCULPT "Sphere 1" GRAB AT 0,-74,255 MOVE 0,-6,0 RADIUS 30
SCULPT "Sphere 1" DRAW AT 0,-82,261 12,-80,259 22,-75,256 RADIUS 7 STRENGTH 0.6 SYMMETRY
SCULPT "Sphere 1" DRAW AT 0,-81,247 10,-80,247 18,-76,249 RADIUS 8 STRENGTH 0.7 SYMMETRY
SCULPT "Sphere 1" CREASE AT 0,-83,254 12,-81,254 24,-75,255 RADIUS 3 STRENGTH 0.4 SYMMETRY
SCULPT "Sphere 1" DRAW INVERT AT 0,-76,238 12,-73,239 RADIUS 8 STRENGTH 0.5 SYMMETRY
SCULPT "Sphere 1" DRAW AT -8,-72,224 8,-72,224 RADIUS 13 STRENGTH 0.8
# cheekbones, and the cheeks filled out below them
SCULPT "Sphere 1" DRAW AT 38,-60,300 52,-48,302 62,-35,300 RADIUS 15 STRENGTH 0.5 SYMMETRY
SCULPT "Sphere 1" INFLATE AT 42,-58,290 52,-50,270 RADIUS 24 STRENGTH 0.6 SYMMETRY
# soften the face before the fine detail
SCULPT "Sphere 1" SMOOTH AT 0,-78,362 30,-72,360 55,-52,355 RADIUS 25 STRENGTH 0.3 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 45,-55,290 55,-40,270 45,-50,240 RADIUS 20 STRENGTH 0.3 SYMMETRY
# eyes: the ball, the upper lid and its fold, the lower lid
SCULPT "Sphere 1" GRAB AT 30,-60,318 MOVE 0,-6,0 RADIUS 11 SYMMETRY
SCULPT "Sphere 1" DRAW AT 19,-62,324 30,-65,327 41,-60,324 RADIUS 5 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" CREASE AT 19,-62,330 30,-64,333 42,-58,329 RADIUS 3 STRENGTH 0.3 SYMMETRY
SCULPT "Sphere 1" DRAW AT 21,-62,312 31,-63,310 40,-59,312 RADIUS 4 STRENGTH 0.7 SYMMETRY
# ears: the flap pulled out and back, the lobe, the bowl pressed in, the rim
SCULPT "Sphere 1" GRAB AT 75,25,312 MOVE 9,7,0 RADIUS 24 SYMMETRY
SCULPT "Sphere 1" GRAB AT 78,22,286 MOVE 3,2,-4 RADIUS 12 SYMMETRY
SCULPT "Sphere 1" GRAB AT 84,14,306 MOVE -7,0,0 RADIUS 10 SYMMETRY
SCULPT "Sphere 1" DRAW AT 80,8,330 84,20,336 88,32,326 89,36,310 86,32,294 RADIUS 5 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" DRAW AT 80,8,330 84,20,336 88,32,326 89,36,310 86,32,294 RADIUS 5 STRENGTH 1 SYMMETRY

# The neck and shoulders: a dome for the shoulder slope, two large balls round the
# chest and the back, cut flat underneath
CYLINDER 54 200 @0,25,100
SCULPT "Cylinder 2" REFINE 8
BOX 440 260 260 @0,10,30
SPHERE 220 @0,15,-295
SPHERE 400 @0,325,-310
SPHERE 400 @0,-305,-310
BOOLEAN "Block 3" INTERSECT "Sphere 4"
BOOLEAN "Block 3" INTERSECT "Sphere 5"
BOOLEAN "Block 3" INTERSECT "Sphere 6"
BOOLEAN "Sphere 1" UNION "Cylinder 2"
BOOLEAN "Sphere 1" UNION "Block 3"
# blend the joins: neck into head, neck into shoulders, the shoulder tips
SCULPT "Sphere 1" SMOOTH AT 0,-25,226 38,-5,235 50,30,255 34,66,268 0,75,270 RADIUS 16 STRENGTH 0.6 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,-25,150 45,-10,140 50,40,140 0,75,148 RADIUS 30 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,-25,150 45,-10,140 50,40,140 0,75,148 RADIUS 30 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 150,-20,75 175,10,45 150,45,75 RADIUS 40 STRENGTH 1 SYMMETRY
# round over the ridges where the shoulder dome meets the chest and the back
SCULPT "Sphere 1" SMOOTH AT 0,-73,127 100,-61,105 160,-41,65 185,-10,35 RADIUS 45 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,94,129 100,81,109 160,58,68 185,30,35 RADIUS 45 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,-73,127 100,-61,105 160,-41,65 185,-10,35 RADIUS 45 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,94,129 100,81,109 160,58,68 185,30,35 RADIUS 45 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,-73,127 100,-61,105 160,-41,65 185,-10,35 RADIUS 35 STRENGTH 1 SYMMETRY
SCULPT "Sphere 1" SMOOTH AT 0,94,129 100,81,109 160,58,68 185,30,35 RADIUS 35 STRENGTH 1 SYMMETRY
# neck muscles from behind the ears to the notch, the notch, the collarbones
SCULPT "Sphere 1" DRAW AT 52,30,262 38,5,210 12,-34,155 RADIUS 11 STRENGTH 0.6 SYMMETRY
SCULPT "Sphere 1" DRAW INVERT AT 0,-48,148 RADIUS 11 STRENGTH 0.8
SCULPT "Sphere 1" DRAW AT 12,-58,140 60,-62,128 120,-52,104 RADIUS 8 STRENGTH 0.7 SYMMETRY
